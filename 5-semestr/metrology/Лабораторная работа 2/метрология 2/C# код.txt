using System;
class Program {
    static void Main() {
        int x = 0;
        int y = 7;
        for (int i = 0; i < 3; i++) {
            if (i % 2 == 0) {
                Console.WriteLine("even " + i);
                x = 10 * x;
                y += 10;
                x--;
            } else if (i == 1) {
                Console.WriteLine("one");
            } else {
                Console.WriteLine("other");
            }
        }
        y = 3;


        if (y < 4) {
            switch (y) {
                case 1:
                    Console.WriteLine("one");
                    break;
                case 2:
                    if (x == 0) {
                        Console.WriteLine("x is zero");
                    } else if (y < 0) {
                        Console.WriteLine("y is negative");
                    }
                    break;
                case 3:
                    Console.WriteLine("3");
                    break;
                default:
                    Console.WriteLine("default");
                    break;
            }
        }

        while (x < 2) {
            x++;
        }

        Console.WriteLine("done");
    }
}
