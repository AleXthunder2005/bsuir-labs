using System;
class Program {
    static void Main() {
        int x = 0;
        int y = 7;
        for (int i = 0; i < 3; i++) {
            if (i % 2 == 0) {
                Console.WriteLine("even " + i);
                x = 10 * x;
                y += 10;
                x--;
            } else if (i == 1) {
                Console.WriteLine("one");
            } else {
                Console.WriteLine("other");
            }
        }
        y = 3;
    }
}
